% tikz/components/box.tex
%
% Reusable component-box style for Figure 1 Panel A.
%
% The signature shape per spec FR-005a:
%   - rectangular corners
%   - NO border on top / left / right
%   - downstream (bottom) edge protrudes as one or more triangles in
%     place of a conventional arrowhead
%
% Loaded by tikz/fig1a_architecture.tex via \input{components/box}.

\usetikzlibrary{positioning, shapes, arrows.meta, calc, fit, backgrounds}

% Default style — fill applied at call site so the same shape can carry
% different palette colours per band / column.
\tikzset{
    dsbox/.style = {
        rectangle,
        rounded corners = 0pt,
        draw            = none,
        inner sep       = 4pt,
        text width      = 28mm,
        align           = center,
        font            = \sffamily\footnotesize,
        minimum height  = 12mm
    },
    dsbox label/.style = {
        font = \sffamily\bfseries\small
    }
}

% \dsbox{<id>}{<at position>}{<fill colour>}{<label>}
%
% Draws the box, fills the body, and adds one (or two — see
% \dsboxtriangles for the variant) triangle(s) protruding from the
% downstream (south) edge. Position the box with \node so the south
% anchor is at the right vertical level for the layer below.
\newcommand{\dsbox}[4]{%
    \node (#1) [dsbox, fill=#3] at #2 {#4};
    \draw[fill=#3, draw=none]
        (#1.south west) --
        ($(#1.south west)!0.4!(#1.south east) - (0, 2.5mm)$) --
        ($(#1.south west)!0.5!(#1.south east) - (0, 0mm)$) --
        ($(#1.south west)!0.6!(#1.south east) - (0, 2.5mm)$) --
        (#1.south east) -- cycle;
}

% \dsboxtriangles{<id>}{<at position>}{<fill colour>}{<label>}{<count>}
%
% Variant emitting <count> triangles along the downstream edge —
% used when a box feeds multiple parallel downstream consumers.
\newcommand{\dsboxtriangles}[5]{%
    \node (#1) [dsbox, fill=#3] at #2 {#4};
    \pgfmathsetmacro{\dsTotal}{#5}
    \foreach \ti in {1, ..., #5} {%
        \pgfmathsetmacro{\dsCenter}{(\ti - 0.5) / \dsTotal}%
        \draw[fill=#3, draw=none]
            ($(#1.south west)!\dsCenter!(#1.south east) - (3mm, 0)$) --
            ($(#1.south west)!\dsCenter!(#1.south east) - (0, 2.5mm)$) --
            ($(#1.south west)!\dsCenter!(#1.south east) + (3mm, 0)$) --
            cycle;
    }
}

% Figure 1 Panel A — OmniPath Metabo architecture diagram.
%
% Milestone A scaffold: compiles cleanly with the right semantic
% content (5 phase bands top→bottom, utils|main|metabo column split
% at the build/database/API bands, dependency arrows, dataset
% sub-boxes inside metabo). Visual polish (final spacing, logo
% placement, arrow styling per FR-005a's downstream-triangle pattern)
% iterates later — the structure is in place.

\documentclass[border = 4mm, tikz]{standalone}

\usepackage{fontspec}
\usepackage{xcolor}
\usepackage{tikz}
\usetikzlibrary{positioning, calc, arrows.meta, backgrounds, fit}

\setsansfont{TeX Gyre Heros}[Scale = MatchLowercase]
\renewcommand{\familydefault}{\sfdefault}

% Palette colours (kept in sync with R/style-palettes.R via the spec).
\definecolor{paletteteal}    {HTML}{006384}
\definecolor{palettemagenta} {HTML}{9F0162}
\definecolor{paletteamber}   {HTML}{FEAF16}
\definecolor{palettelime}    {HTML}{BBCC33}
\definecolor{palettecoral}   {HTML}{EA6572}
\definecolor{palettegreen}   {HTML}{009E73}
\definecolor{palettesky}     {HTML}{99DDFF}
\definecolor{palettepink}    {HTML}{D03293}
\definecolor{paletteunknown} {HTML}{BEBEBE}

\begin{document}
\begin{tikzpicture}[
    node distance = 4mm,
    every node/.style = {font = \sffamily\footnotesize, align = center}
]

% Helpers -----------------------------------------------------------------
\tikzset{
    band/.style       = {rectangle, minimum height = 14mm, inner sep = 2mm,
                          font = \sffamily\footnotesize, align = center,
                          rounded corners = 1pt, draw = none},
    bandlabel/.style  = {anchor = east, font = \sffamily\small\bfseries,
                          text width = 26mm, align = right},
    downstream/.style = {-{Latex[length = 2.5mm, width = 2mm]},
                          thick, color = black!85},
    lateral/.style    = {-{Latex[length = 2.5mm, width = 2mm]},
                          thick, color = black!70},
    shortcut/.style   = {-{Latex[length = 2.5mm, width = 2mm]},
                          thick, dashed, color = paletteunknown!80!black},
    planned/.style    = {-{Latex[length = 2.5mm, width = 2mm]},
                          thick, densely dashdotted, color = paletteamber!80!black}
}

% Band 1 — Original resources (full width) --------------------------------
\node[band, fill = paletteteal!25, minimum width = 110mm]
    (resources) at (0, 0)
    {\textbf{$\sim$200 upstream resources}\\
     {\scriptsize signaling, metabolic, chemical, ontologies}};
\node[bandlabel] at ($(resources.west) + (-2mm, 0)$)
    {Original\\resources};

% Band 2 — Download \& processing (full width) ----------------------------
\node[band, fill = palettesky!50, minimum width = 110mm, below = 5mm of resources]
    (pyresources)
    {\textbf{omnipath-resources}\\
     {\scriptsize (formerly pypath; download + parsing)}};
\node[bandlabel] at ($(pyresources.west) + (-2mm, 0)$)
    {Download \&\\processing};

% Band 3 — Database build (utils | main | metabo) --------------------------
\node[band, fill = paletteamber!40, minimum width = 32mm, below = 5mm of pyresources.south west, anchor = north west]
    (utils_build) {\textbf{omnipath-utils}\\{\scriptsize ID resolution}};
\node[band, fill = paletteteal!40, minimum width = 32mm, right = 5mm of utils_build]
    (main_build) {\textbf{omnipath-build}\\{\scriptsize ID + CV harmonization}};
\node[band, fill = palettemagenta!30, minimum width = 32mm, right = 5mm of main_build]
    (metabo_build) {\textbf{omnipath-metabo}\\{\scriptsize standalone build}};
\node[bandlabel] at ($(utils_build.west) + (-2mm, 0)$)
    {Database\\build};

% Band 4 — Database (utils DB | main DB + metabo sub-boxes) ---------------
\node[band, fill = paletteamber!20, minimum width = 32mm, below = 5mm of utils_build]
    (utils_db) {\textbf{utils DB}};
\node[band, fill = paletteteal!20, minimum width = 32mm, right = 5mm of utils_db]
    (main_db) {\textbf{OmniPath DB}\\{\scriptsize Postgres}};
\node[band, fill = palettemagenta!20, minimum width = 32mm, right = 5mm of main_db, inner sep = 2mm]
    (metabo_panel) {};

% Inside the metabo column, three sub-boxes.
\node[fill = palettemagenta!35, minimum width = 28mm, minimum height = 4mm, font = \scriptsize, inner sep = 1mm, anchor = north]
    (metalinks) at ($(metabo_panel.north) + (0, -1mm)$)
    {MetalinksDB};
\node[fill = palettelime!50, minimum width = 28mm, minimum height = 4mm, font = \scriptsize, inner sep = 1mm, below = 0.5mm of metalinks]
    (cosmos) {COSMOS PKN};
\node[fill = palettecoral!40, minimum width = 28mm, minimum height = 4mm, font = \scriptsize, inner sep = 1mm, below = 0.5mm of cosmos]
    (cheminfo) {Cheminformatics (RDKit)};
\node[bandlabel] at ($(utils_db.west) + (-2mm, 0)$)
    {Database};

% Band 5 — Web API (3 separate APIs) --------------------------------------
\node[band, fill = paletteamber!30, minimum width = 32mm, below = 5mm of utils_db]
    (utils_api) {\textbf{utils.omnipathdb.org}};
\node[band, fill = paletteteal!30, minimum width = 32mm, right = 5mm of utils_api]
    (main_api) {\textbf{omnipathdb.org}\\{\scriptsize HTTP API + web app}};
\node[band, fill = palettemagenta!30, minimum width = 32mm, right = 5mm of main_api]
    (metabo_api) {\textbf{metabo.omnipathdb.org}};
\node[bandlabel] at ($(utils_api.west) + (-2mm, 0)$)
    {Web API\\\& app};

% Band 6 — Clients (full width) -------------------------------------------
\node[band, fill = palettesky!40, minimum width = 25mm, below = 5mm of utils_api.south west, anchor = north west]
    (omnipathr) {\textbf{OmnipathR}\\{\scriptsize R / Bioconductor}};
\node[band, fill = palettelime!50, minimum width = 25mm, right = 3mm of omnipathr]
    (client) {\textbf{omnipath-client}\\{\scriptsize Python}};
\node[band, fill = palettecoral!40, minimum width = 25mm, right = 3mm of client]
    (annnet) {\textbf{annnet}\\{\scriptsize network DS}};
\node[band, fill = paletteamber!50, minimum width = 25mm, right = 3mm of annnet]
    (cytoscape) {\textbf{Cytoscape}};
\node[bandlabel] at ($(omnipathr.west) + (-2mm, 0)$) {Clients};

% Connections ------------------------------------------------------------
\draw[downstream] (resources)   -- (pyresources);
\draw[downstream] (pyresources.south west) ++(20mm, 0) -- (utils_build);
\draw[downstream] (pyresources)              -- (main_build);
\draw[shortcut]   (pyresources.south east) ++(-20mm, 0) -- (metabo_build);

\draw[lateral]    (utils_build) -- (main_build)
                  node[midway, above, font = \tiny] {ID res.};
\draw[lateral]    (main_build)  -- (metabo_build);
\draw[planned]    (utils_build.south east) ++(0, -4mm)
                  to[bend right = 8] ($(metabo_build.south west) + (0, -4mm)$);

\draw[downstream] (utils_build)  -- (utils_db);
\draw[downstream] (main_build)   -- (main_db);
\draw[downstream] (metabo_build) -- (metabo_panel.north);

\draw[downstream] (utils_db)     -- (utils_api);
\draw[downstream] (main_db)      -- (main_api);
\draw[downstream] (cheminfo)     -- (metabo_api);

% MetalinksDB origin: dashed cross-column arrow from main DB.
\draw[->, thick, color = palettemagenta!70!black, shorten >= 1mm, shorten <= 1mm]
    (main_db.east) -- (metalinks.west)
    node[midway, above, font = \tiny, color = black] {materialized};

\draw[downstream] (utils_api)  -- (omnipathr);
\draw[downstream] (main_api)   -- (client);
\draw[downstream] (metabo_api) -- (annnet);
\draw[downstream] (annnet)     -- (cytoscape)
                  node[midway, above, font = \tiny] {opens in};

% utils → omnipath-client bypass (utils Python package is directly importable).
\draw[downstream, color = paletteamber!80!black]
    (utils_api.west) to[out = -90, in = 90] ($(omnipathr.north) + (-10mm, 0)$);

\end{tikzpicture}
\end{document}

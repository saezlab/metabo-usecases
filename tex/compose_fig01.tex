% Invoked by figures/fig01-overview/build.R via the rebuild driver →
% R/compose/mixed_source.R::compose_mixed_source(mode = "pdf", ...).
%
% Panel input paths are written relative to the repo root; the rebuild
% driver runs xelatex with -output-directory pointed at the figure's

    width=\singlecolwidth]{panelE.pdf}

\includepdf[pages=1, fitpaper=false, pagecommand={\panellabel{F}},
    width=\singlecolwidth]{panelF.pdf}

\end{document}
